    public function __construct() {
        if ( defined( 'BBL_OPENAI_API_KEY' ) ) {
            $this->openai_api_key = BBL_OPENAI_API_KEY;
        } else {
            $this->openai_api_key = getenv( 'OPENAI_API_KEY' );
        }

        add_action( 'rest_api_init', array( $this, 'register_routes' ) );
    }

    public function register_routes() {
        register_rest_route(
            'bbl/v1',
            '/answer',
            array(
                'methods'             => 'POST',
                'callback'            => array( $this, 'handle_answer_request' ),
                'permission_callback' => '__return_true',
            )
        );
    }

    public function handle_answer_request( WP_REST_Request $request ) {
        $params   = $request->get_json_params();
        $question = isset( $params['question'] ) ? sanitize_text_field( $params['question'] ) : '';
        $city     = isset( $params['city'] ) ? sanitize_text_field( $params['city'] ) : '';

        if ( empty( $question ) ) {
            return new WP_REST_Response(
                array( 'error' => 'Missing question' ),
                400
            );
        }

        // Try ZIP first
        $zip = $this->extract_zip_code( $question );
        if ( $zip ) {
            $zip_city = $this->map_zip_to_city_slug( $zip );
            if ( $zip_city ) {
                $city = $zip_city;
            }
        }

        // Default anchor if none resolved
        if ( empty( $city ) ) {
            $city = 'lynnwood-wa';
        }

        $category_slug = $this->infer_category_from_question( $question );
        $city_cluster  = $this->get_city_cluster_for_radius( $city );
        $city_name     = $this->map_city_slug_to_name( $city );
        $businesses    = $this->get_matching_businesses( $city_cluster, $category_slug );

        $context = array(
            'anchor_city_slug' => $city,
            'anchor_city_name' => $city_name,
            'cluster'          => $city_cluster,
            'category_slug'    => $category_slug,
            'question'         => $question,
            'businesses'       => $businesses,
            'category_url'     => $this->guess_category_url( $city, $category_slug ),
        );

        $answer_html = $this->generate_answer_with_ai( $context );

        if ( is_wp_error( $answer_html ) ) {
            return new WP_REST_Response(
                array( 'error' => $answer_html->get_error_message() ),
                500
            );
        }

        return new WP_REST_Response(
            array(
                'answer_html' => $answer_html,
                'raw_results' => $businesses,
            ),
            200
        );
    }

    private function extract_zip_code( $question ) {
        if ( preg_match( '/\b\d{5}\b/', $question, $matches ) ) {
            return $matches[0];
        }
        return null;
    }

    private function infer_category_from_question( $question ) {
        $q = strtolower( $question );

        $map = array(
            'restaurants' => array( 'food', 'restaurant', 'eat', 'dinner', 'lunch', 'brunch', 'pizza', 'burger', 'taco', 'tacos', 'sushi', 'date night' ),
            'nightlife'   => array( 'bar', 'nightlife', 'cocktail', 'beer', 'pub', 'late night' ),
            'wellness'    => array( 'yoga', 'gym', 'workout', 'fitness', 'spa', 'massage', 'wellness', 'pilates' ),
            'home-services' => array( 'plumber', 'electrician', 'roof', 'roofer', 'hvac', 'contractor', 'remodel', 'repair' ),
            'local'       => array( 'coffee', 'shop', 'boutique', 'local store', 'gift', 'bookstore' ),
        );

        foreach ( $map as $slug => $keywords ) {
            foreach ( $keywords as $kw ) {
                if ( strpos( $q, $kw ) !== false ) {
                    return $slug;
                }
            }
        }

        return 'local';
    }

    private function get_matching_businesses( $city_slugs, $category_slug ) {
        if ( ! is_array( $city_slugs ) ) {
            $city_slugs = array( $city_slugs );
        }

        $args = array(
            'post_type'      => 'business',
            'posts_per_page' => 5,
            'post_status'    => 'publish',
            'tax_query'      => array(
                'relation' => 'AND',
                array(
                    'taxonomy' => 'bbl_city',
                    'field'    => 'slug',
                    'terms'    => $city_slugs,
                ),
                array(
                    'taxonomy' => 'bbl_category',
                    'field'    => 'slug',
                    'terms'    => $category_slug,
                ),
            ),
        );

        $query   = new WP_Query( $args );
        $results = array();

        if ( $query->have_posts() ) {
            foreach ( $query->posts as $post ) {
                $results[] = array(
                    'title'   => get_the_title( $post ),
                    'url'     => get_permalink( $post ),
                    'phone'   => get_post_meta( $post->ID, 'bbl_phone', true ),
                    'address' => get_post_meta( $post->ID, 'bbl_address', true ),
                );
            }
        }

        wp_reset_postdata();
        return $results;
    }

    private function guess_category_url( $anchor_city_slug, $category_slug ) {
        return home_url( '/' . $anchor_city_slug . '/' . $category_slug . '/' );
    }

    private function generate_answer_with_ai( $context ) {
        if ( ! $this->openai_api_key ) {
            return new WP_Error( 'missing_api_key', 'OpenAI API key not configured.' );
        }

        $endpoint = 'https://api.openai.com/v1/chat/completions';

        $business_list_text = '';
        foreach ( $context['businesses'] as $b ) {
            $business_list_text .= sprintf(
                "- %s (%s) — %s\n",
                $b['title'],
                $b['phone'],
                $b['url']
            );
        }

        $cluster_str = implode( ', ', $context['cluster'] );

        $system_prompt = sprintf(
            "You are a friendly local guide for %s and the nearby area. ".
            "You speak like a real person who lives there: relaxed, clear, and helpful. ".
            "You NEVER sound like a robot or an AI assistant. ".
            "You NEVER invent businesses that are not in the list provided. ".
            "You keep answers short (2–4 sentences), specific, and easy to skim. ".
            "You usually recommend the anchor city first, then nearby cities within about 10–20 minutes if it makes sense. ".
            "If options are limited, you’re honest about it but still encouraging, and you invite the visitor to explore the category page link.",
            $context['anchor_city_name']
        );

        $user_prompt = "Visitor question: {$context['question']}\n\n".
            "Anchor city: {$context['anchor_city_name']} (slug: {$context['anchor_city_slug']})\n".
            "Cluster city slugs (approx 10–25 minute radius): {$cluster_str}\n".
            "Inferred category: {$context['category_slug']}\n".
            "Category page URL: {$context['category_url']}\n\n".
            "Here are businesses you can recommend:\n".
            $business_list_text.
            "\n\nWrite the answer as if you’re a local talking to a neighbor. ".
            "Use natural, conversational language. ".
            "Mention 2–3 specific businesses by name with each name hyperlinked to its URL. ".
            "Start with spots in the anchor city when possible. ".
            "You can say things like 'If you’re in the mood for…' or 'If you don’t mind a quick drive…'. ".
            "End with one short line like: 'You can see more local options here' linking to the category page.\n";

        $body = array(
            'model'    => 'gpt-4.1-mini',
            'messages' => array(
                array( 'role' => 'system', 'content' => $system_prompt ),
                array( 'role' => 'user',   'content' => $user_prompt ),
            ),
            'temperature' => 0.55,
        );

        $response = wp_remote_post(
            $endpoint,
            array(
                'headers' => array(
                    'Content-Type'  => 'application/json',
                    'Authorization' => 'Bearer ' . $this->openai_api_key,
                ),
                'body'    => wp_json_encode( $body ),
                'timeout' => 30,
            )
        );

        if ( is_wp_error( $response ) ) {
            return $response;
        }

        $code = wp_remote_retrieve_response_code( $response );
        if ( $code < 200 || $code >= 300 ) {
            return new WP_Error( 'openai_http_error', 'OpenAI API error: HTTP ' . $code );
        }

        $data = json_decode( wp_remote_retrieve_body( $response ), true );

        if ( ! isset( $data['choices'][0]['message']['content'] ) ) {
            return new WP_Error( 'openai_bad_response', 'Unexpected OpenAI response.' );
        }

        return trim( $data['choices'][0]['message']['content'] );
    }
